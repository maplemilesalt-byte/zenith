cmake_minimum_required(VERSION 3.20)
project(Zenith VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(zenith src/main.cpp src/window.cpp src/renderer.cpp src/elf_loader.cpp src/memory/guest_memory.cpp src/cpu/cpu.cpp src/appx/appx_package.cpp src/appx/appx_loader.cpp src/input/input_state.cpp)
target_include_directories(zenith PRIVATE src)
find_package(X11 REQUIRED)
find_package(ZLIB REQUIRED)
target_link_libraries(zenith PRIVATE X11::X11 ZLIB::ZLIB)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(zenith PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_program(ZENITH_CLANG clang REQUIRED)
find_program(ZENITH_LLD ld.lld REQUIRED)
find_program(ZENITH_ZIP zip REQUIRED)
set(ZENITH_HELLO_OBJ ${CMAKE_CURRENT_BINARY_DIR}/helloworld.o)
set(ZENITH_HELLO_ELF ${CMAKE_CURRENT_BINARY_DIR}/helloworld.elf)
add_custom_command(OUTPUT ${ZENITH_HELLO_OBJ} COMMAND ${ZENITH_CLANG} --target=x86_64-unknown-linux-gnu -c ${CMAKE_CURRENT_SOURCE_DIR}/examples/helloworld.S -o ${ZENITH_HELLO_OBJ} DEPENDS examples/helloworld.S COMMENT "Assembling x86-64 Zenith guest helloworld" VERBATIM)
add_custom_command(OUTPUT ${ZENITH_HELLO_ELF} COMMAND ${ZENITH_LLD} -m elf_x86_64 -T ${CMAKE_CURRENT_SOURCE_DIR}/examples/helloworld.ld -o ${ZENITH_HELLO_ELF} ${ZENITH_HELLO_OBJ} DEPENDS ${ZENITH_HELLO_OBJ} examples/helloworld.ld COMMENT "Linking x86-64 Zenith guest helloworld.elf" VERBATIM)
add_custom_target(zenith_helloworld ALL DEPENDS ${ZENITH_HELLO_ELF})
add_dependencies(zenith zenith_helloworld)
add_custom_target(helloworld DEPENDS ${ZENITH_HELLO_ELF})
set(ZENITH_GRAPHICS_OBJ ${CMAKE_CURRENT_BINARY_DIR}/graphics.o)
set(ZENITH_GRAPHICS_ELF ${CMAKE_CURRENT_BINARY_DIR}/graphics.elf)
add_custom_command(OUTPUT ${ZENITH_GRAPHICS_OBJ} COMMAND ${ZENITH_CLANG} --target=x86_64-unknown-linux-gnu -c ${CMAKE_CURRENT_SOURCE_DIR}/examples/graphics.S -o ${ZENITH_GRAPHICS_OBJ} DEPENDS examples/graphics.S COMMENT "Assembling x86-64 Zenith graphical guest" VERBATIM)
add_custom_command(OUTPUT ${ZENITH_GRAPHICS_ELF} COMMAND ${ZENITH_LLD} -m elf_x86_64 -T ${CMAKE_CURRENT_SOURCE_DIR}/examples/helloworld.ld -o ${ZENITH_GRAPHICS_ELF} ${ZENITH_GRAPHICS_OBJ} DEPENDS ${ZENITH_GRAPHICS_OBJ} examples/helloworld.ld COMMENT "Linking x86-64 Zenith graphics.elf" VERBATIM)
add_custom_target(zenith_graphics ALL DEPENDS ${ZENITH_GRAPHICS_ELF})
add_dependencies(zenith zenith_graphics)
add_custom_target(graphics DEPENDS ${ZENITH_GRAPHICS_ELF})
set(ZENITH_APPX_OBJ ${CMAKE_CURRENT_BINARY_DIR}/ZenithPlatformer.o)
set(ZENITH_APPX_ELF ${CMAKE_CURRENT_BINARY_DIR}/ZenithPlatformer.elf)
set(ZENITH_APPX ${CMAKE_CURRENT_BINARY_DIR}/ZenithPlatformer.appx)
add_custom_command(OUTPUT ${ZENITH_APPX_OBJ} COMMAND ${ZENITH_CLANG} --target=x86_64-unknown-linux-gnu -c ${CMAKE_CURRENT_SOURCE_DIR}/examples/appx_platformer.S -o ${ZENITH_APPX_OBJ} DEPENDS examples/appx_platformer.S COMMENT "Assembling x86-64 Zenith APPX platformer" VERBATIM)
add_custom_command(OUTPUT ${ZENITH_APPX_ELF} COMMAND ${ZENITH_LLD} -m elf_x86_64 -T ${CMAKE_CURRENT_SOURCE_DIR}/examples/helloworld.ld -o ${ZENITH_APPX_ELF} ${ZENITH_APPX_OBJ} DEPENDS ${ZENITH_APPX_OBJ} examples/helloworld.ld COMMENT "Linking Zenith APPX platformer ELF" VERBATIM)
add_custom_command(OUTPUT ${ZENITH_APPX} COMMAND ${ZENITH_ZIP} -q -j ${ZENITH_APPX} ${CMAKE_CURRENT_SOURCE_DIR}/examples/AppxManifest.xml ${ZENITH_APPX_ELF} DEPENDS ${ZENITH_APPX_ELF} examples/AppxManifest.xml COMMENT "Packaging Zenith Platformer APPX" VERBATIM)
add_custom_target(zenith_appx_platformer ALL DEPENDS ${ZENITH_APPX})
add_custom_target(appx_platformer DEPENDS ${ZENITH_APPX})
set(ZENITH_INPUT_TEST_OBJ ${CMAKE_CURRENT_BINARY_DIR}/ZenithInputTest.o)
set(ZENITH_INPUT_TEST_ELF ${CMAKE_CURRENT_BINARY_DIR}/ZenithInputTest.elf)
add_custom_command(OUTPUT ${ZENITH_INPUT_TEST_OBJ} COMMAND ${ZENITH_CLANG} --target=x86_64-unknown-linux-gnu -c ${CMAKE_CURRENT_SOURCE_DIR}/examples/input_test.S -o ${ZENITH_INPUT_TEST_OBJ} DEPENDS examples/input_test.S COMMENT "Assembling Zenith input test guest" VERBATIM)
add_custom_command(OUTPUT ${ZENITH_INPUT_TEST_ELF} COMMAND ${ZENITH_LLD} -m elf_x86_64 -T ${CMAKE_CURRENT_SOURCE_DIR}/examples/helloworld.ld -o ${ZENITH_INPUT_TEST_ELF} ${ZENITH_INPUT_TEST_OBJ} DEPENDS ${ZENITH_INPUT_TEST_OBJ} examples/helloworld.ld COMMENT "Linking ZenithInputTest.elf" VERBATIM)
add_custom_target(zenith_input_test ALL DEPENDS ${ZENITH_INPUT_TEST_ELF})
add_dependencies(zenith zenith_input_test)
add_custom_target(input_test DEPENDS ${ZENITH_INPUT_TEST_ELF})
enable_testing()
add_executable(zenith_memory_tests tests/guest_memory_tests.cpp src/memory/guest_memory.cpp)
target_include_directories(zenith_memory_tests PRIVATE src)
add_test(NAME zenith_memory_tests COMMAND zenith_memory_tests)
add_executable(zenith_cpu_tests tests/cpu_tests.cpp src/cpu/cpu.cpp src/memory/guest_memory.cpp)
target_include_directories(zenith_cpu_tests PRIVATE src)
add_test(NAME zenith_cpu_tests COMMAND zenith_cpu_tests)
add_executable(zenith_input_tests tests/input_tests.cpp src/input/input_state.cpp)
target_include_directories(zenith_input_tests PRIVATE src)
add_test(NAME zenith_input_tests COMMAND zenith_input_tests)
