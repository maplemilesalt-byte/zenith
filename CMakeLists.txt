cmake_minimum_required(VERSION 3.20)

option(ZENITH_ENABLE_WAYLAND "Build the Wayland window backend when dependencies are available" ON)
if(ZENITH_ENABLE_WAYLAND)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(WAYLAND QUIET wayland-client)
    pkg_check_modules(WAYLAND_PROTOCOLS QUIET wayland-protocols)
  endif()
  find_program(WAYLAND_SCANNER wayland-scanner)
  if(WAYLAND_FOUND AND WAYLAND_PROTOCOLS_FOUND AND WAYLAND_SCANNER)
    execute_process(COMMAND ${PKG_CONFIG_EXECUTABLE} --variable=pkgdatadir wayland-protocols OUTPUT_VARIABLE WAYLAND_PROTOCOL_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
    set(XDG_XML ${WAYLAND_PROTOCOL_DIR}/stable/xdg-shell/xdg-shell.xml)
    set(XDG_HEADER ${CMAKE_CURRENT_BINARY_DIR}/xdg-shell-client-protocol.h)
    set(XDG_CODE ${CMAKE_CURRENT_BINARY_DIR}/xdg-shell-protocol.c)
    add_custom_command(OUTPUT ${XDG_HEADER} ${XDG_CODE}
      COMMAND ${WAYLAND_SCANNER} client-header ${XDG_XML} ${XDG_HEADER}
      COMMAND ${WAYLAND_SCANNER} private-code ${XDG_XML} ${XDG_CODE}
      DEPENDS ${XDG_XML})
    add_library(zenith_wayland_protocol STATIC ${XDG_CODE} ${XDG_HEADER})
    target_include_directories(zenith_wayland_protocol PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(zenith_wayland_protocol PUBLIC ${WAYLAND_LIBRARIES})
    target_include_directories(zenith_wayland_protocol PUBLIC ${WAYLAND_INCLUDE_DIRS})
    target_compile_options(zenith_wayland_protocol PRIVATE ${WAYLAND_CFLAGS_OTHER})
    target_sources(zenith PRIVATE src/wayland_window.cpp ${XDG_HEADER})
    target_include_directories(zenith PRIVATE ${WAYLAND_INCLUDE_DIRS} ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(zenith PRIVATE zenith_wayland_protocol ${WAYLAND_LIBRARIES})
    target_compile_definitions(zenith PRIVATE ZENITH_HAS_WAYLAND=1)
    message(STATUS "Zenith Wayland backend: enabled")
  else()
    message(STATUS "Zenith Wayland backend: disabled (missing wayland-client, wayland-protocols, or wayland-scanner)")
  endif()
endif()
